a=input("nhap danh sach cac so nguyen (cach nhau boi dau cach): ")
danh_sach_so=[]
for i in a.split():
    danh_sach_so=[int(i) for i in a.split()]
tong= sum(danh_sach_so)
tb= tong/len(danh_sach_so)
lon_nhat= max(danh_sach_so)
print(f"tong: {tong}") 
print(f"trung binh cong: {tb}")
print(f"so lon nhat: {lon_nhat}")
